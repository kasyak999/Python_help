📦 СИСТЕМНОЕ ОБНОВЛЕНИЕ
yay -Syu             # Обновить всю систему и AUR

📥 УСТАНОВКА И ПОИСК
yay -S <пакет>        # Установить пакет
yay -Ss <запрос>      # Найти пакет в репозиториях и AUR
yay -Qs <запрос>      # Найти пакет среди установленных
yay -Qi <пакет>       # Подробная информация о пакете

🗑 УДАЛЕНИЕ И ОЧИСТКА
yay -Rns <пакет>      # Удалить пакет вместе с конфигами и его сиротами
yay -Qdt            # Показать неиспользуемые зависимости (сирот)
yay -Rns $(yay -Qdtq)         #  Автоматически удалить всех сирот из системы
yay -Q                      # Покажет вообще всё: программы, библиотеки, драйверы, зависимости и т.д.
sudo paccache -r      # Безопасная очистка кэша (оставить 3 версии пакетов)
yay -Sc               # Полная очистка кэша пакетов и сборщика AUR

🛠 ДИАГНОСТИКА
pacman -Qo /путь      # Узнать, какая программа создала этот файл
less /var/log/pacman.log # Посмотреть историю установки/обновления пакетов

yay -Qe (-q без версий)
показать мои установленые пакеты

ncdu ~
Показать что занимает много места

df -h
информация о диске

yay -Qqe > pkglist.txt
список установленых пакетов

pactree -r xf86-video-amdgpu     # Посмотреть зависиомсти пакета
